! =========================
! BASIC CONFIGURATION

!

hostname ACCESS-SW2
no ip domain-lookup

!
! =========================
! VLAN DATABASE
! =========================
!

vlan 10
 name SALES

vlan 20
 name IT

vlan 30
 name SERVERS

vlan 40
 name GUEST

vlan 50
 name VOICE

vlan 99
 name MANAGEMENT

vlan 999
 name NATIVE-BLACKHOLE

! =========================
! GUEST-PC1
! =========================
!

interface FastEthernet0/1
 description GUEST-PC1
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

!
! =========================
! DHCP/DNS SERVER
! =========================
!

interface FastEthernet0/2
 description SRV-DHCP-DNS
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

!
! =========================
! WEB SERVER
! =========================
!

interface FastEthernet0/3
 description SRV-WEB
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

!
! =========================
! UNUSED PORTS
! =========================
!

interface range FastEthernet0/4 - 22
 description UNUSED_PORT
 switchport mode access
 switchport access vlan 999
 shutdown

! UPLINK TO CORE-SW2

 description TRUNK_TO_CORE-SW2
 switchport mode trunk
